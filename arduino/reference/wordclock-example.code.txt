/* ========== Hardware / layout ========== */
#define DATA_PIN 5
#define NUM_LEDS 114

RTC_DS3231 rtc;
CRGB leds[NUM_LEDS];
CRGB targetLeds[NUM_LEDS];

WebServer server(80);
DNSServer dnsServer;
Preferences prefs;

const byte DNS_PORT = 53;

/* ========== Defaults ========== */
const uint8_t DEFAULT_DAY_BRI_PERCENT = 70;
const uint8_t DEFAULT_NIGHT_BRI_PERCENT = 20;
const uint8_t DEFAULT_NIGHT_START = 22;
const uint8_t DEFAULT_NIGHT_END = 6;
const CRGB DEFAULT_COLOR = CRGB(255, 225, 200);

/* ========== Stored settings (runtime) ========== */
uint8_t dayBrightPercent = DEFAULT_DAY_BRI_PERCENT;
uint8_t nightBrightPercent = DEFAULT_NIGHT_BRI_PERCENT;
uint8_t nightStartHour = DEFAULT_NIGHT_START;
uint8_t nightEndHour = DEFAULT_NIGHT_END;
CRGB WORD_COLOR = DEFAULT_COLOR;

/* Preference keys */
const char *P_NS = "nightStart";
const char *P_NE = "nightEnd";
const char *P_DB = "dayB";
const char *P_NB = "nightB";
const char *P_R = "rc";
const char *P_G = "gc";
const char *P_B = "bc";

/* ========== WIFI / AP status ========== */
bool apActive = false;
bool clientConnected = false;
unsigned long apStartTime = 0;

/* Pulsing */
int pulseValue = 10;
int pulseDir = 8;

/* Status LED indices */
const uint8_t WIFI_LEDS[4] = { 15, 16, 17, 18 };

/* ========== DEBUG ========== */
#define DEBUG_SERIAL 1   // pon a 0 para desactivar todos los prints de debug
String debugWords = "";  // se reconstruye en cada buildTimeTarget()

inline void logW(const char *name) {
#if DEBUG_SERIAL
  debugWords += name;
  debugWords += " ";
#endif
}

const uint8_t DOS[] = { 19, 20, 21 };

const uint8_t DOCE[] = { 66, 67, 68, 69 };

const uint8_t MIN_LEDS[4] = { 110, 111, 112, 113 };

/* ========== Helpers ========== */

inline bool isWifiLed(uint8_t idx) {
  for (uint8_t i = 0; i < 4; i++)
    if (WIFI_LEDS[i] == idx) return true;
  return false;
}

void clearTarget() {
  fill_solid(targetLeds, NUM_LEDS, CRGB::Black);
}

void setW(const uint8_t *arr, uint8_t len, const char *name = nullptr) {
  for (uint8_t i = 0; i < len; i++) {
    uint8_t idx = arr[i];
    if (idx < NUM_LEDS && !isWifiLed(idx)) targetLeds[idx] = WORD_COLOR;
  }
  if (name) logW(name);
}

void setMinuteDotsTarget(uint8_t count) {
  for (uint8_t i = 0; i < 4; i++) {
    uint8_t idx = MIN_LEDS[i];
    if (idx < NUM_LEDS) targetLeds[idx] = (i < count) ? WORD_COLOR : CRGB::Black;
  }
}

/* ========== Debug dump ========== */
void printDebugStatus(int hourVal, int minuteVal) {
#if DEBUG_SERIAL
  DateTime now = rtc.now();
  Serial.println(F("---------------------------------"));
  Serial.printf("RTC ahora: %02d:%02d:%02d  %02d/%02d/%04d\n",
                now.hour(), now.minute(), now.second(),
                now.day(), now.month(), now.year());
  Serial.printf("Hora solicitada para render: %02d:%02d\n", hourVal, minuteVal);
  Serial.print(F("Frase generada: "));
  Serial.println(debugWords);

  Serial.print(F("LEDs encendidos (sin WIFI_LEDS): "));
  bool first = true;
  for (int i = 0; i < NUM_LEDS; i++) {
    if (targetLeds[i].r != 0 || targetLeds[i].g != 0 || targetLeds[i].b != 0) {
      if (!first) Serial.print(", ");
      Serial.print(i);
      first = false;
    }
  }
  Serial.println();
  Serial.println(F("---------------------------------"));
#endif
}

/* ========== Crossfade ========== */

void crossFade(uint8_t steps = 45, uint16_t delayMs = 18) {
  for (uint8_t s = 0; s <= steps; s++) {
    uint8_t amt = (255 * s) / steps;
    for (int i = 0; i < NUM_LEDS; i++) {
      leds[i] = blend(leds[i], targetLeds[i], amt);
    }
    FastLED.show();
    delay(delayMs);
  }
}

/* ========== Time -> target builder ========== */

void hour(uint8_t h) {
  switch (h) {
    case 1: setW(UNA, 3, "UNA"); break;
    case 2: setW(DOS, 3, "DOS"); break;
    case 3: setW(TRES, 4, "TRES"); break;
    case 4: setW(CUATRO, 6, "CUATRO"); break;
    case 5: setW(CINCO_H, 5, "CINCO_H"); break;
    case 6: setW(SEIS, 4, "SEIS"); break;
    case 7: setW(SIETE, 5, "SIETE"); break;
    case 8: setW(OCHO, 4, "OCHO"); break;
    case 9: setW(NUEVE, 5, "NUEVE"); break;
    case 10: setW(DIEZ_H, 4, "DIEZ_H"); break;
    case 11: setW(ONCE, 4, "ONCE"); break;
    case 12: setW(DOCE, 4, "DOCE"); break;
  }
}

bool isNightHour(uint8_t hour) {
  if (nightStartHour == nightEndHour) return false;
  if (nightStartHour < nightEndHour) {
    return (hour >= nightStartHour && hour < nightEndHour);
  } else {
    return (hour >= nightStartHour || hour < nightEndHour);
  }
}

void setSpanishIntro(uint8_t displayHour) {
  if (displayHour == 1) {
    setW(ES, 2, "ES");
    setW(LA, 2, "LA");
  } else {
    setW(SON, 3, "SON");
    setW(LAS, 3, "LAS");
  }
}

void buildTimeTarget(int hourVal, int minuteVal) {

  debugWords = "";  // reset de la frase de debug en cada construccion

  uint8_t effectivePercent = isNightHour(hourVal) ? nightBrightPercent : dayBrightPercent;
  uint8_t brightnessVal = map(effectivePercent, 0, 100, 0, 255);
  FastLED.setBrightness(brightnessVal);

  clearTarget();

  uint8_t rounded = (minuteVal / 5) * 5;
  uint8_t extra = minuteVal % 5;

  uint8_t h = hourVal % 12;
  if (h == 0) h = 12;

  uint8_t nextHour = (h % 12) + 1;

  switch (rounded) {

    case 0:
      setSpanishIntro(h);
      hour(h);
      break;

    case 5:
      setSpanishIntro(h);
      hour(h);
      setW(Y, 1, "Y");
      setW(CINCO, 5, "CINCO");
      break;

    case 10:
      setSpanishIntro(h);
      hour(h);
      setW(Y, 1, "Y");
      setW(DIEZ, 4, "DIEZ");
      break;

    case 15:
      setSpanishIntro(h);
      hour(h);
      setW(Y, 1, "Y");
      setW(CUARTO, 6, "CUARTO");
      break;

    case 20:
      setSpanishIntro(h);
      hour(h);
      setW(Y, 1, "Y");
      setW(VEINTE, 6, "VEINTE");
      break;

    case 25:
      setSpanishIntro(h);
      hour(h);
      setW(Y, 1, "Y");
      setW(VEINTICINCO, 11, "VEINTICINCO");
      break;

    case 30:
      setSpanishIntro(h);
      hour(h);
      setW(Y, 1, "Y");
      setW(MEDIA, 5, "MEDIA");
      break;

    case 35:
      setSpanishIntro(nextHour);
      hour(nextHour);
      setW(MENOS, 5, "MENOS");
      setW(VEINTICINCO, 11, "VEINTICINCO");
      break;

    case 40:
      setSpanishIntro(nextHour);
      hour(nextHour);
      setW(MENOS, 5, "MENOS");
      setW(VEINTE, 6, "VEINTE");
      break;

    case 45:
      setSpanishIntro(nextHour);
      hour(nextHour);
      setW(MENOS, 5, "MENOS");
      setW(CUARTO, 6, "CUARTO");
      break;

    case 50:
      setSpanishIntro(nextHour);
      hour(nextHour);
      setW(MENOS, 5, "MENOS");
      setW(DIEZ, 4, "DIEZ");
      break;

    case 55:
      setSpanishIntro(nextHour);
      hour(nextHour);
      setW(MENOS, 5, "MENOS");
      setW(CINCO, 5, "CINCO");
      break;
  }

  setMinuteDotsTarget(extra);

  printDebugStatus(hourVal, minuteVal);
}

void handleRoot() {

}

void handleGenerate204() {
  server.send(200, "text/plain", "OK");
}

void handleSave() {
  if (server.hasArg("c")) {
    String c = server.arg("c");
    if (c.length() == 7 && c[0] == '#') {
      long v = strtol(c.substring(1).c_str(), NULL, 16);
      WORD_COLOR = CRGB((v >> 16) & 255, (v >> 8) & 255, v & 255);
    }
  }
  if (server.hasArg("day")) {
    int d = server.arg("day").toInt();
    if (d < 0) d = 0;
    if (d > 100) d = 100;
    dayBrightPercent = d;
  }
  if (server.hasArg("night")) {
    int n = server.arg("night").toInt();
    if (n < 0) n = 0;
    if (n > 100) n = 100;
    nightBrightPercent = n;
  }
  if (server.hasArg("ns")) {
    int ns = server.arg("ns").toInt();
    if (ns < 0) ns = 0;
    if (ns > 23) ns = 23;   // <-- FIX: antes era "= 0"
    nightStartHour = ns;
  }
  if (server.hasArg("ne")) {
    int ne = server.arg("ne").toInt();
    if (ne < 0) ne = 0;
    if (ne > 23) ne = 23;   // <-- FIX: antes era "= 0"
    nightEndHour = ne;
  }

  bool gotDate = false;
  int y = 0, mo = 0, da = 0, hh = 0, mm = 0;
  if (server.hasArg("y") && server.hasArg("mo") && server.hasArg("da") && server.hasArg("h") && server.hasArg("mi")) {
    y = server.arg("y").toInt();
    mo = server.arg("mo").toInt();
    da = server.arg("da").toInt();
    hh = server.arg("h").toInt();
    mm = server.arg("mi").toInt();
    if (y >= 2000 && y <= 2099 && mo >= 1 && mo <= 12 && da >= 1 && da <= 31 && hh >= 0 && hh <= 23 && mm >= 0 && mm <= 59) {
      gotDate = true;
    }
  }

  prefs.begin("wortuhr", false);
  prefs.putUChar(P_DB, dayBrightPercent);
  prefs.putUChar(P_NB, nightBrightPercent);
  prefs.putUChar(P_NS, nightStartHour);
  prefs.putUChar(P_NE, nightEndHour);
  prefs.putUChar(P_R, WORD_COLOR.r);
  prefs.putUChar(P_G, WORD_COLOR.g);
  prefs.putUChar(P_B, WORD_COLOR.b);
  prefs.end();

  if (gotDate) {
    rtc.adjust(DateTime(y, mo, da, hh, mm, 0));
  }

#if DEBUG_SERIAL
  Serial.println(F("=== Configuracion guardada ==="));
  Serial.printf("Color: #%02X%02X%02X\n", WORD_COLOR.r, WORD_COLOR.g, WORD_COLOR.b);
  Serial.printf("Brillo dia/noche: %d%% / %d%%\n", dayBrightPercent, nightBrightPercent);
  Serial.printf("Noche: %02d a %02d\n", nightStartHour, nightEndHour);
  if (gotDate) Serial.printf("RTC ajustado a: %04d-%02d-%02d %02d:%02d\n", y, mo, da, hh, mm);
#endif

  server.send(200, "text/html",
              "<html><body style='background:#111;color:white;text-align:center;font-family:Arial'>"
              "<h2>Guardado!</h2><p>El reloj se reinicia...</p></body></html>");

  delay(900);
  dnsServer.stop();
  WiFi.softAPdisconnect(true);
  delay(200);
  ESP.restart();
}

void handleSetTime() {
  if (server.hasArg("y") && server.hasArg("mo") && server.hasArg("da") && server.hasArg("h") && server.hasArg("mi") && server.hasArg("s")) {

    int y = server.arg("y").toInt();
    int mo = server.arg("mo").toInt();
    int da = server.arg("da").toInt();
    int h = server.arg("h").toInt();
    int mi = server.arg("mi").toInt();
    int s = server.arg("s").toInt();

    rtc.adjust(DateTime(y, mo, da, h, mi, s));

#if DEBUG_SERIAL
    Serial.printf("RTC fijado via /settime: %04d-%02d-%02d %02d:%02d:%02d\n", y, mo, da, h, mi, s);
#endif

    server.send(200, "text/plain", "OK");
    return;
  }
  server.send(400, "text/plain", "BAD");
}

void handleNotFound() {
  IPAddress apIP = WiFi.softAPIP();
  server.sendHeader("Location", String("http://") + apIP.toString() + "/", true);
  server.send(302, "text/plain", "");
}

/* ========== Preferences load ========== */
void loadSettings() {
  prefs.begin("wortuhr", true);
  dayBrightPercent = prefs.getUChar(P_DB, DEFAULT_DAY_BRI_PERCENT);
  nightBrightPercent = prefs.getUChar(P_NB, DEFAULT_NIGHT_BRI_PERCENT);
  nightStartHour = prefs.getUChar(P_NS, DEFAULT_NIGHT_START);
  nightEndHour = prefs.getUChar(P_NE, DEFAULT_NIGHT_END);
  uint8_t r = prefs.getUChar(P_R, DEFAULT_COLOR.r);
  uint8_t g = prefs.getUChar(P_G, DEFAULT_COLOR.g);
  uint8_t b = prefs.getUChar(P_B, DEFAULT_COLOR.b);
  WORD_COLOR = CRGB(r, g, b);
  prefs.end();

#if DEBUG_SERIAL
  Serial.println(F("=== Configuracion cargada de NVS ==="));
  Serial.printf("Color: #%02X%02X%02X\n", WORD_COLOR.r, WORD_COLOR.g, WORD_COLOR.b);
  Serial.printf("Brillo dia/noche: %d%% / %d%%\n", dayBrightPercent, nightBrightPercent);
  Serial.printf("Noche: %02d a %02d\n", nightStartHour, nightEndHour);
#endif
}

/* ========== Start AP & DNS & Webserver ========== */
void startAP() {
  WiFi.mode(WIFI_AP);
  WiFi.softAP("WORDCLOCK-SETUP");

  apStartTime = millis();
  apActive = true;
  clientConnected = false;
  pulseValue = 10;
  pulseDir = 8;

  IPAddress apIP = WiFi.softAPIP();
  dnsServer.start(DNS_PORT, "*", apIP);

  server.on("/", handleRoot);
  server.on("/generate_204", handleGenerate204);
  server.on("/fwlink", handleGenerate204);
  server.on("/save", handleSave);
  server.on("/settime", handleSetTime);
  server.onNotFound(handleNotFound);
  server.begin();

#if DEBUG_SERIAL
  Serial.print(F("AP iniciado. IP: "));
  Serial.println(apIP);
#endif
}

/* ========== Update status LEDs (apActive / clientConnected) ========== */
void applyWifiStatusVisuals() {
  if (apActive) {
    int stations = WiFi.softAPgetStationNum();
    if (stations > 0) {
      for (uint8_t i = 0; i < 4; i++) {
        uint8_t idx = WIFI_LEDS[i];
        if (idx < NUM_LEDS) leds[idx] = CRGB(0, 120, 255);
      }
      return;
    } else {
      pulseValue += pulseDir;
      if (pulseValue >= 200 || pulseValue <= 6) pulseDir = -pulseDir;
      uint8_t pv = constrain(pulseValue, 0, 255);
      for (uint8_t i = 0; i < 4; i++) {
        uint8_t idx = WIFI_LEDS[i];
        if (idx < NUM_LEDS) leds[idx] = CRGB(0, 0, pv);
      }
      return;
    }
  } else {
    for (uint8_t i = 0; i < 4; i++) {
      uint8_t idx = WIFI_LEDS[i];
      if (idx < NUM_LEDS) leds[idx] = CRGB::Black;
    }
  }
}

/* ========== Core time show wrapper ========== */
int lastMinute = -1;

void showTimeAndFade(int h, int m) {
  buildTimeTarget(h, m);
  crossFade();
  applyWifiStatusVisuals();
  FastLED.show();
}

void startAnimation() {
  FastLED.setBrightness(120);

  fill_solid(leds, NUM_LEDS, CRGB::Black);
  FastLED.show();

  uint8_t order[NUM_LEDS];
  for (int i = 0; i < NUM_LEDS; i++) order[i] = i;

  for (int i = NUM_LEDS - 1; i > 0; i--) {
    int j = random(i + 1);
    uint8_t t = order[i];
    order[i] = order[j];
    order[j] = t;
  }

  for (int i = 0; i < NUM_LEDS; i++) {
    leds[order[i]] = CHSV(random8(), 255, 200);
    FastLED.show();
    delay(35);
  }

  delay(500);

  for (int b = 200; b >= 0; b -= 5) {
    FastLED.setBrightness(b);
    FastLED.show();
    delay(20);
  }

  FastLED.setBrightness(255);
  fill_solid(leds, NUM_LEDS, CRGB::Black);
  FastLED.show();
}

/* ========== Setup & loop ========== */

void setup() {
  Serial.begin(115200);
  delay(50);

  FastLED.addLeds<WS2812, DATA_PIN, GRB>(leds, NUM_LEDS);
  fill_solid(leds, NUM_LEDS, CRGB::Black);
  fill_solid(targetLeds, NUM_LEDS, CRGB::Black);

  Wire.begin();
  if (!rtc.begin()) {
    Serial.println(F("ERROR: RTC DS3231 no detectado en el bus I2C!"));
  }

  if (rtc.lostPower()) {
    Serial.println(F("RTC perdio la alimentacion, ajustando a hora de compilacion."));
    rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
  }

  loadSettings();

  startAnimation();

  startAP();

  DateTime now = rtc.now();
  lastMinute = now.minute();
  buildTimeTarget(now.hour(), now.minute());
  crossFade(18, 12);

  applyWifiStatusVisuals();
  FastLED.show();
}

void loop() {
  dnsServer.processNextRequest();
  server.handleClient();

  if (apActive) {
    int stations = WiFi.softAPgetStationNum();
    if (stations > 0) {
      clientConnected = true;
    }

    if (!clientConnected && (millis() - apStartTime > 60000UL)) {
      dnsServer.stop();
      WiFi.softAPdisconnect(true);
      apActive = false;
      clientConnected = false;
      for (uint8_t i = 0; i < 4; i++) {
        if (WIFI_LEDS[i] < NUM_LEDS) leds[WIFI_LEDS[i]] = CRGB::Black;
      }
      FastLED.show();
#if DEBUG_SERIAL
      Serial.println(F("AP apagado: timeout sin clientes."));
#endif
    }

    if (clientConnected && (WiFi.softAPgetStationNum() == 0) && (millis() - apStartTime > 500)) {
      dnsServer.stop();
      WiFi.softAPdisconnect(true);
      apActive = false;
      clientConnected = false;
      for (uint8_t i = 0; i < 4; i++) {
        if (WIFI_LEDS[i] < NUM_LEDS) leds[WIFI_LEDS[i]] = CRGB::Black;
      }
      FastLED.show();
#if DEBUG_SERIAL
      Serial.println(F("AP apagado: cliente se desconecto."));
#endif
    }
  }

  applyWifiStatusVisuals();

  DateTime now = rtc.now();
  if (now.minute() != lastMinute) {
    lastMinute = now.minute();
    showTimeAndFade(now.hour(), now.minute());
  } else {
    FastLED.show();
  }

  delay(80);
}
